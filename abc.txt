-- Update ownership
ALTER TABLE IF EXISTS public.email_queue OWNER to rap_user;

-- Add any missing columns (e.g. attachments) in case they weren't fully synced
ALTER TABLE public.email_queue ADD COLUMN IF NOT EXISTS attachments jsonb;
ALTER TABLE public.email_queue ADD COLUMN IF NOT EXISTS status text NOT NULL DEFAULT 'QUEUED';

-- Index: ix_email_queue_consultant_id
CREATE INDEX IF NOT EXISTS ix_email_queue_consultant_id
    ON public.email_queue USING btree
    (consultant_id ASC NULLS LAST)
    TABLESPACE pg_default;

-- Index: ix_email_queue_id
CREATE INDEX IF NOT EXISTS ix_email_queue_id
    ON public.email_queue USING btree
    (id ASC NULLS LAST)
    TABLESPACE pg_default;

-- Index: ix_email_queue_requirement_id
CREATE INDEX IF NOT EXISTS ix_email_queue_requirement_id
    ON public.email_queue USING btree
    (requirement_id ASC NULLS LAST)
    TABLESPACE pg_default;

    name VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    consultant_id BIGINT NOT NULL,
    match_score NUMERIC(5,2),
    matching_info JSONB,
    match_reasoning TEXT,
    status TEXT NOT NULL DEFAULT 'PENDING',
    created_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_job_matches_requirements FOREIGN KEY (requirement_id) REFERENCES requirements (id) ON DELETE CASCADE,
    CONSTRAINT fk_job_matches_consultants FOREIGN KEY (consultant_id) REFERENCES consultants (id) ON DELETE CASCADE,
    CONSTRAINT uq_job_match_req_cons UNIQUE (requirement_id, consultant_id)
);

-- 5. Database View for pre-joined Pending Job Matches
CREATE OR REPLACE VIEW v_pending_job_matches AS
SELECT 
    jm.id AS id,
    jm.requirement_id AS requirement_id,
    r.role AS requirement_title,
    COALESCE(r.client, r.vendor) AS requirement_company,
    jm.consultant_id AS consultant_id,
    c.full_name AS consultant_name,
    c.email AS consultant_email,
    jm.match_score AS match_score,
    jm.match_reasoning AS match_reasoning,
    jm.status AS status,
    jm.created_at AS created_at
FROM job_matches jm
JOIN requirements r ON jm.requirement_id = r.id
JOIN consultants c ON jm.consultant_id = c.id;

-- 1. Database View for System & User Reporting Stats
CREATE OR REPLACE VIEW v_user_reporting_stats AS
SELECT 
    u.id AS user_id,
    u.full_name AS user_name,
    u.role AS user_role,
    COALESCE(app_counts.applications_sent, 0) AS applications_sent,
    COALESCE(eq_counts.emails_sent, 0) AS emails_sent
FROM users u
LEFT JOIN (
    SELECT recruiter_id, COUNT(*) AS applications_sent
    FROM applications
    WHERE status = 'SENT' AND recruiter_id IS NOT NULL
    GROUP BY recruiter_id
) app_counts ON u.id = app_counts.recruiter_id
LEFT JOIN (
    SELECT sent_by_user_id, COUNT(*) AS emails_sent
    FROM email_queue
    WHERE status = 'SENT' AND sent_by_user_id IS NOT NULL
    GROUP BY sent_by_user_id
) eq_counts ON u.id = eq_counts.sent_by_user_id;

    eq.content AS content,

    eq.status_text AS status_text,

    eq.updated_at AS updated_at
FROM email_queue eq
LEFT JOIN consultants c ON eq.consultant_id = c.id
LEFT JOIN requirements r ON eq.requirement_id = r.id;

UPDATE users
SET is_authorized = CASE
    WHEN role IN ('ADMIN', 'RECRUITER') THEN TRUE
    WHEN id IN (
        SELECT c.user_id 
        FROM consultants c
        JOIN consultant_email_tokens t ON c.id = t.consultant_id
    ) THEN TRUE
    ELSE FALSE
END;

UPDATE consultants
SET gmail_connected = CASE
    WHEN id IN (SELECT consultant_id FROM consultant_email_tokens) THEN TRUE
    ELSE FALSE
END;

DELETE FROM requirement_consultant_matches

    SELECT id FROM requirements WHERE raw_email_id IN (
        SELECT id FROM gmail_emails ORDER BY id DESC LIMIT 100
    )
);

-- 2. Delete the actual requirements generated from the top 100 emails
DELETE FROM requirements 
WHERE raw_email_id IN (
    SELECT id FROM gmail_emails ORDER BY id DESC LIMIT 100
);

-- 3. Reset the state of the top 100 emails so the cron sees them as "new"
UPDATE gmail_emails 
SET status_desc = 'Pending', category = NULL, reason = NULL, processed = false 
WHERE id IN (
    SELECT id FROM gmail_emails ORDER BY id DESC LIMIT 100
);

COMMIT;

-- Fix received_date for recently created requirements that are invisible in the UI:
UPDATE requirements 
SET received_date = CAST(gmail_emails.date AS timestamp with time zone)
FROM gmail_emails 
WHERE requirements.raw_email_id = gmail_emails.id 
  AND requirements.received_date IS NULL;




-- Reset top 100 emails for reprocessing
BEGIN;
DELETE FROM requirement_consultant_matches WHERE requirement_id IN (SELECT id FROM requirements WHERE raw_email_id IN (SELECT id FROM gmail_emails ORDER BY id DESC LIMIT 100));
DELETE FROM requirements WHERE raw_email_id IN (SELECT id FROM gmail_emails ORDER BY id DESC LIMIT 100);

-- database_views_and_optimizations.sql
-- =========================================================================================
-- RAP Platform Database Views and Performance Optimizations
-- =========================================================================================
-- This script contains:
--   1. Reusable SQL Views replacing complex multi-table joins across endpoints.
--   2. Targeted B-tree and composite indexes to eliminate full-table scans.
--
-- Safe to run multiple times: all view creations use CREATE OR REPLACE VIEW / DROP VIEW IF EXISTS,
-- and all indexes use CREATE INDEX IF NOT EXISTS.
-- =========================================================================================

-- -----------------------------------------------------------------------------------------
-- VIEW 1: v_pending_matches_active
-- -----------------------------------------------------------------------------------------
-- Purpose: Consolidates the 4-table join (RequirementConsultantMatch + Requirement +
--          Consultant + User) and authorization gating used on the Pending Applications page,
--          Consultant Matched view, and Recruiter Roster matching views.
--          Includes `has_sent_application` subquery check.
-- -----------------------------------------------------------------------------------------
CREATE OR REPLACE VIEW v_pending_matches_active AS
SELECT
    rcm.id,
    rcm.requirement_id,
    r.role AS requirement_title,
    COALESCE(r.client, r.vendor) AS requirement_company,
    r.vendor_email AS requirement_vendor_email,
    r.raw_email_id,
    rcm.consultant_id,
    c.full_name AS consultant_name,
    c.email AS consultant_email,
    rcm.match_score,
    rcm.score_breakdown,
    rcm.match_reason,
    rcm.status,
    rcm.resume_status,
    rcm.created_at,
    rcm.updated_at,
    c.user_id,
    EXISTS (
        SELECT 1 FROM applications a
        WHERE a.requirement_id = rcm.requirement_id
          AND a.consultant_id = rcm.consultant_id
          AND a.status = 'SENT'
    ) AS has_sent_application
FROM requirement_consultant_matches rcm
JOIN requirements r ON rcm.requirement_id = r.id
JOIN consultants c ON rcm.consultant_id = c.id
JOIN users u ON u.id = c.user_id
WHERE c.status = 'ACTIVE'
  AND u.is_authorized = true;


-- -----------------------------------------------------------------------------------------
-- VIEW 2: v_application_tracker_details
-- -----------------------------------------------------------------------------------------
-- Purpose: Consolidates Application, Consultant, Requirement, and Recruiter (User) tables
--          for recruiter and consultant application tracker pages.
-- -----------------------------------------------------------------------------------------
DROP VIEW IF EXISTS v_application_tracker_details;
CREATE VIEW v_application_tracker_details AS
SELECT
    a.id,
    a.consultant_id,
    c.full_name AS consultant_name,
    c.email AS consultant_email,
    a.requirement_id,
    r.role AS requirement_title,
    COALESCE(r.client, r.vendor) AS company_name,
    r.client,
    r.vendor,
    a.recruiter_id,
    u.full_name AS recruiter_name,
    u.role AS recruiter_role,
    a.status,
    a.vendor_email,
    a.cc_email,
    a.email_subject,
    a.gmail_message_id,
    a.generated_resume_id,
    a.resume_attachment_path,
    (a.generated_resume_id IS NOT NULL OR a.resume_attachment_path IS NOT NULL) AS resume_available,
    a.sent_at,
    a.applied_at,
    a.created_at,
    a.updated_at,
    a.error_message
FROM applications a
JOIN consultants c ON a.consultant_id = c.id
JOIN requirements r ON a.requirement_id = r.id
LEFT JOIN users u ON a.recruiter_id = u.id;


-- -----------------------------------------------------------------------------------------
-- VIEW 3: v_email_queue_details
-- -----------------------------------------------------------------------------------------
-- Purpose: Pre-joins EmailQueue with Consultant, Requirement, and Sender User details
--          to eliminate repetitive outer joins across email queue listings and searches.
-- -----------------------------------------------------------------------------------------
DROP VIEW IF EXISTS v_email_queue_details;
CREATE VIEW v_email_queue_details AS
SELECT
    eq.id,
    eq.consultant_id,
    c.full_name AS consultant_name,
    c.email AS consultant_email,
    eq.requirement_id,
    r.role AS requirement_title,
    r.client AS requirement_client,
    r.vendor AS requirement_vendor,
    eq.from_email,
    eq.to_email,
    eq.cc_email,
    eq.subject,
    eq.content,
    eq.html_content,
    eq.attachments,
    eq.status,
    eq.status_text,
    eq.sent_by_user_id,
    u.full_name AS sent_by_name,
    u.role AS sent_by_role,
    eq.scheduled_at,
    eq.created_at,
    eq.updated_at
FROM email_queue eq
LEFT JOIN consultants c ON eq.consultant_id = c.id
LEFT JOIN requirements r ON eq.requirement_id = r.id
LEFT JOIN users u ON eq.sent_by_user_id = u.id;


-- -----------------------------------------------------------------------------------------
-- VIEW 4: v_user_reporting_stats
-- -----------------------------------------------------------------------------------------
-- Purpose: Aggregates sent applications and email queue submissions grouped per user
--          for the reporting and admin dashboard overview.
-- -----------------------------------------------------------------------------------------
CREATE OR REPLACE VIEW v_user_reporting_stats AS
SELECT
    u.id AS user_id,
    u.full_name AS user_name,
    u.role AS user_role,
    COALESCE(app_counts.applications_sent, 0::bigint) AS applications_sent,
    COALESCE(eq_counts.emails_sent, 0::bigint) AS emails_sent
FROM users u
LEFT JOIN (
    SELECT applications.recruiter_id, count(*) AS applications_sent
    FROM applications
    WHERE applications.status = 'SENT' AND applications.recruiter_id IS NOT NULL
    GROUP BY applications.recruiter_id
) app_counts ON u.id = app_counts.recruiter_id
LEFT JOIN (
    SELECT email_queue.sent_by_user_id, count(*) AS emails_sent
    FROM email_queue
    WHERE email_queue.status = 'SENT' AND email_queue.sent_by_user_id IS NOT NULL
    GROUP BY email_queue.sent_by_user_id
) eq_counts ON u.id = eq_counts.sent_by_user_id;


-- -----------------------------------------------------------------------------------------
-- Performance Indexes Supporting the Views and High-Traffic Queries
-- -----------------------------------------------------------------------------------------

-- 1. Covering composite index on requirement_consultant_matches for requirement lookups
CREATE INDEX IF NOT EXISTS ix_rcm_req_status_id
ON requirement_consultant_matches (requirement_id, status, id);

-- 2. Partial index for fast MATCHING lookups per consultant & requirement ordered by creation
CREATE INDEX IF NOT EXISTS ix_rcm_matching_cons_req_created
ON requirement_consultant_matches (consultant_id, requirement_id, created_at DESC)
WHERE status = 'MATCHING';

-- 3. Composite and sorted indexes on applications for instant sent_at pagination
CREATE INDEX IF NOT EXISTS ix_applications_sent_at_desc
ON applications (sent_at DESC NULLS LAST);

CREATE INDEX IF NOT EXISTS ix_applications_sent_status
ON applications (status, sent_at DESC NULLS LAST);

-- 4. Status and ID index on requirements for quick index-only group counts
CREATE INDEX IF NOT EXISTS ix_requirements_status_id
ON requirements (status, id);


-- 5. Full name index for fast user list ordering
CREATE INDEX IF NOT EXISTS ix_users_full_name_asc ON users (full_name ASC);


-- 6. Reporting time-window indexes
CREATE INDEX IF NOT EXISTS ix_emails_parse_received ON emails (parse_status, received_at);
CREATE INDEX IF NOT EXISTS ix_applications_sent_range ON applications (sent_at) WHERE status = 'SENT';
CREATE INDEX IF NOT EXISTS ix_applications_created_range ON applications (created_at);
CREATE INDEX IF NOT EXISTS ix_email_queue_created_range ON email_queue (created_at) WHERE status = 'SENT';

-- 7. Notification unread count index
CREATE INDEX IF NOT EXISTS ix_notifications_user_unread ON notifications (user_id, is_read);
